cmake_minimum_required(VERSION 3.24)
project(RenderSchedule LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120-real)
set(UPSTREAM ${CMAKE_CURRENT_SOURCE_DIR}/offline_inbox/renderer/diff-gaussian-rasterization)
add_library(native_core STATIC ${UPSTREAM}/cuda_rasterizer/forward.cu ${UPSTREAM}/cuda_rasterizer/backward.cu ${UPSTREAM}/cuda_rasterizer/rasterizer_impl.cu)
target_include_directories(native_core PUBLIC ${UPSTREAM}/cuda_rasterizer ${UPSTREAM}/third_party/glm)
target_compile_options(native_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
add_executable(native_renderer src/native_main.cu)
target_link_libraries(native_renderer PRIVATE native_core)

target_compile_options(native_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--pre-include=cstdint>)

add_library(scheduled_core STATIC src/core/forward.cu src/core/backward.cu src/core/rasterizer_impl.cu src/scheduler.cu)
target_include_directories(scheduled_core PUBLIC src/core src ${UPSTREAM}/third_party/glm)
target_compile_options(scheduled_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3> $<$<COMPILE_LANGUAGE:CUDA>:--pre-include=cstdint>)
add_executable(scheduled_renderer src/scheduled_main.cu)
target_link_libraries(scheduled_renderer PRIVATE scheduled_core)
add_executable(benchmark src/benchmark.cu)
target_link_libraries(benchmark PRIVATE scheduled_core)
