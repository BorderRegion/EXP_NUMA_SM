# /etc/systemd/system/t10-quiet-fan.service
[Unit]
Description=Tesla T10 temperature based CHA_FAN quiet curve
Wants=t10-quiet-power.service
After=t10-quiet-power.service
StartLimitIntervalSec=0

[Service]
Type=notify
NotifyAccess=main
ExecStartPre=/usr/sbin/modprobe nct6775
ExecStart=/usr/bin/python3 -u /usr/local/sbin/t10_quiet_fan.py
ExecStopPost=/usr/bin/python3 -u /usr/local/sbin/t10_quiet_fan.py --restore
Restart=always
RestartSec=3
WatchdogSec=8
WatchdogSignal=SIGKILL
TimeoutStartSec=30
TimeoutStopSec=6

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/t10-quiet-power.service
[Unit]
Description=Tesla T10 quiet power profile at 125 W
Wants=nvidia-persistenced.service
After=systemd-modules-load.service nvidia-persistenced.service
StartLimitIntervalSec=0

[Service]
Type=oneshot
ExecStart=/usr/bin/nvidia-smi -i GPU-8cfbc08d-0090-bc29-db9a-b2a2f7261ca5 -pm 1

RemainAfterExit=yes
Restart=on-failure
RestartSec=15
TimeoutStartSec=45

[Install]
WantedBy=multi-user.target
